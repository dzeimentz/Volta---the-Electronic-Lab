# Tests fuer Schaltungsmodell und Dienste (Qt Test, docs/testing.md)
find_package(Qt6 REQUIRED COMPONENTS Test)

function(volta_qt_test name source)
    qt_add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE volta_model Qt6::Test)
    set_target_properties(${name} PROPERTIES WIN32_EXECUTABLE OFF) # Konsolenprogramm: Testausgabe sichtbar
    target_include_directories(${name} PRIVATE "${VOLTA_GENERATED_DIR}") # volta_paths.hpp (VOLTA_CONTENT_DIR)
    add_test(NAME ${name} COMMAND ${name})
    # Qt-Bibliotheken fuer den Testlauf auffindbar machen (Windows: PATH)
    if(WIN32)
        get_target_property(_qmake Qt6::qmake IMPORTED_LOCATION)
        get_filename_component(_qtbin "${_qmake}" DIRECTORY)
        set_tests_properties(${name} PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qtbin}")
    endif()
endfunction()

volta_qt_test(volta_model_tests test_model.cpp)
volta_qt_test(volta_content_smoke_tests test_content_smoke.cpp)
volta_qt_test(volta_bridge_tests test_bridge.cpp)
volta_qt_test(volta_session_tests test_session.cpp)

# Oberflaechentest: echte QML-Oberflaeche ohne sichtbares Fenster (offscreen)
